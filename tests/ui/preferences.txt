# Preferences, Reset to Defaults and the inspector overlay:
#   NodeLab.exe tests\ui\interact.nlproj --script tests\ui\preferences.txt
# Changes Invert's Factor and resets it from the node menu, switches the Inspector to the
# overlay in Edit > Preferences, picks the Blender theme, then drags the overlay's slider.
settle
drag "node:Invert.Factor" -50 0
settle
wait 5
click "node:Invert" 1
wait 5
click "Reset to Defaults"
wait 5
shot build/smoke/prefs_reset.png
click "Edit"
wait 3
click "Preferences..."
wait 5
expect "Preferences/Resolution Scale"
click "Preferences/Overlay"
wait 5
click "Preferences/Themes"
wait 3
click "Preferences/Theme"
wait 3
click "Blender"
wait 3
shot build/smoke/prefs_theme.png
click "Preferences/#CLOSE"
wait 5
settle
click "node:Invert"
wait 5
drag "Inspector/Factor" -100 0
wheel 3
settle
shot build/smoke/prefs_overlay.png
quit
