# Library Grid: copy photos to build\smoke\lib (see library.txt), then
#   Refractory.exe build\smoke\lib --script tests\ui\grid.txt
# G opens the grid; a click selects, Ctrl+click adds, a star click rates, 2 rates the
# selection, P picks it; a double-click opens a photo and leaves the grid.
settle
wait 20
settle
move 800 500
key g
settle
wait 30
settle
shot build/smoke/grid1.png
# The fourth star of the second photo, then select it and Ctrl+click the third
clickat "Library Grid/card#2" 0.578 0.93
click "Library Grid/card#2"
ctrl on
click "Library Grid/card#3"
ctrl off
key 2
key p
settle
shot build/smoke/grid2.png
# Double-click the fourth: it opens, and the grid closes
doubleclick "Library Grid/card#4"
settle
wait 30
settle
shot build/smoke/grid3.png
quit
